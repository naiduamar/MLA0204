
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import LabelEncoder
from sklearn.naive_bayes import GaussianNB
from sklearn.metrics import accuracy_score, classification_report

# Create sample bank loan dataset
data = {
    'Income': ['High', 'Low', 'High', 'Medium', 'Low',
               'High', 'Medium', 'Low', 'High', 'Medium',
               'Low', 'High', 'Medium', 'Low'],
    'Credit_History': ['Good', 'Bad', 'Good', 'Good', 'Bad',
                       'Good', 'Bad', 'Bad', 'Good', 'Good',
                       'Bad', 'Good', 'Good', 'Bad'],
    'Employment': ['Yes', 'No', 'Yes', 'Yes', 'No',
                   'Yes', 'No', 'No', 'Yes', 'Yes',
                   'No', 'Yes', 'Yes', 'No'],
    'Loan_Amount': ['High', 'Low', 'Medium', 'High', 'Low',
                    'Medium', 'Low', 'High', 'High', 'Medium',
                    'Low', 'Medium', 'High', 'Low'],
    'Loan_Status': ['Approved', 'Rejected', 'Approved',
                    'Approved', 'Rejected', 'Approved',
                    'Rejected', 'Rejected', 'Approved',
                    'Approved', 'Rejected', 'Approved',
                    'Approved', 'Rejected']
}

df = pd.DataFrame(data)

print("Bank Loan Dataset:\n")
print(df)

# Convert categorical values into numerical values
encoders = {}

for column in df.columns:
    le = LabelEncoder()
    df[column] = le.fit_transform(df[column])
    encoders[column] = le

# Separate input and output
X = df.drop('Loan_Status', axis=1)
y = df['Loan_Status']

# Split dataset
X_train, X_test, y_train, y_test = train_test_split(
    X, y, test_size=0.25, random_state=42, stratify=y
)

# Create Naive Bayes model
model = GaussianNB()

# Train model
model.fit(X_train, y_train)

print("\nNaive Bayes model trained successfully.")

# Predict test data
y_pred = model.predict(X_test)

print("\nActual Loan Status:")
print(encoders['Loan_Status'].inverse_transform(y_test))

print("\nPredicted Loan Status:")
print(encoders['Loan_Status'].inverse_transform(y_pred))

# Calculate accuracy
accuracy = accuracy_score(y_test, y_pred)

print("\nAccuracy:", accuracy * 100, "%")

print("\nClassification Report:")
print(classification_report(
    y_test, y_pred,
    target_names=encoders['Loan_Status'].classes_,
    zero_division=0
))

# Predict for a new applicant
new_applicant = pd.DataFrame(
    [['High', 'Good', 'Yes', 'Medium']],
    columns=['Income', 'Credit_History', 'Employment', 'Loan_Amount']
)

# Encode new applicant using existing encoders
for column in new_applicant.columns:
    new_applicant[column] = encoders[column].transform(
        new_applicant[column]
    )

prediction = model.predict(new_applicant)

result = encoders['Loan_Status'].inverse_transform(prediction)

print("\nNew Applicant Loan Prediction:", result[0])
